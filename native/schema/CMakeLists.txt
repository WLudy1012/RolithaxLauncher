cmake_minimum_required(VERSION 3.22)
project(rolithax_schema VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
option(ROLITHAX_SCHEMA_CLI "Build JSON-lines CLI" ON)
option(ROLITHAX_SCHEMA_JNI "Include JVM JNI adapter for host tests" OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "vendor/lua/*.c")
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac|linit|lcorolib|ldblib|liolib|loadlib|loslib)\\.c$")
add_library(rolithax_schema SHARED src/schema.cpp vendor/tinyxml2.cpp ${LUA_SOURCES})
if(ROLITHAX_SCHEMA_JNI)
  find_path(SCHEMA_JNI_INCLUDE jni.h HINTS "${JAVA_HOME}/include" "$ENV{JAVA_HOME}/include" REQUIRED)
  find_path(SCHEMA_JNI_PLATFORM_INCLUDE jni_md.h HINTS "${SCHEMA_JNI_INCLUDE}" PATH_SUFFIXES win32 linux darwin REQUIRED)
  target_sources(rolithax_schema PRIVATE ../../app/src/main/cpp/schema_jni.cpp)
  target_include_directories(rolithax_schema PRIVATE ${SCHEMA_JNI_INCLUDE} ${SCHEMA_JNI_PLATFORM_INCLUDE})
endif()
find_package(Threads REQUIRED)
target_link_libraries(rolithax_schema PRIVATE Threads::Threads)
target_compile_features(rolithax_schema PRIVATE cxx_std_17)
target_compile_definitions(rolithax_schema PRIVATE ROLITHAX_SCHEMA_BUILD)
target_include_directories(rolithax_schema PUBLIC include PRIVATE vendor vendor/lua)
set_target_properties(rolithax_schema PROPERTIES CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_custom_command(TARGET rolithax_schema POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:rolithax_schema>/licenses"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE.md" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JSON-LICENSE.MIT" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/TINYXML2-LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/LUA-LICENSE.html" "$<TARGET_FILE_DIR:rolithax_schema>/licenses"
  VERBATIM)
if(MSVC)
  target_compile_options(rolithax_schema PRIVATE /utf-8 /W3)
else()
  target_compile_options(rolithax_schema PRIVATE -Wall -Wextra -Wno-unused-parameter)
  target_link_libraries(rolithax_schema PRIVATE m)
endif()
if(ROLITHAX_SCHEMA_CLI)
  add_executable(rolithax-schema src/cli.cpp)
  target_compile_features(rolithax-schema PRIVATE cxx_std_17)
  target_link_libraries(rolithax-schema PRIVATE rolithax_schema)
  if(APPLE)
    set_target_properties(rolithax-schema PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path;@loader_path/../lib")
  else()
    set_target_properties(rolithax-schema PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib")
  endif()
endif()
if(BUILD_TESTING)
  add_executable(schema-tests tests/schema_tests.cpp)
  target_compile_features(schema-tests PRIVATE cxx_std_17)
  target_include_directories(schema-tests PRIVATE vendor)
  target_link_libraries(schema-tests PRIVATE rolithax_schema Threads::Threads)
  if(MSVC)
    target_compile_options(schema-tests PRIVATE /utf-8)
  endif()
  add_test(NAME schema-tests COMMAND schema-tests)
  set_tests_properties(schema-tests PROPERTIES TIMEOUT 30)
  if(ROLITHAX_SCHEMA_JNI)
    find_program(SCHEMA_JAVAC javac HINTS "${JAVA_HOME}/bin" "$ENV{JAVA_HOME}/bin" REQUIRED)
    find_program(SCHEMA_JAVA java HINTS "${JAVA_HOME}/bin" "$ENV{JAVA_HOME}/bin" REQUIRED)
    set(JNI_TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/tests/jni/com/wludy/rolithax/launcher/shared/console/SchemaNativeBridge.java")
    set(JNI_TEST_CLASS "${CMAKE_CURRENT_BINARY_DIR}/jni-test/com/wludy/rolithax/launcher/shared/console/SchemaNativeBridge.class")
    add_custom_command(OUTPUT "${JNI_TEST_CLASS}" COMMAND "${SCHEMA_JAVAC}" -encoding UTF-8 -d "${CMAKE_CURRENT_BINARY_DIR}/jni-test" "${JNI_TEST_SOURCE}" DEPENDS "${JNI_TEST_SOURCE}" VERBATIM)
    add_custom_target(schema-jni-test ALL DEPENDS "${JNI_TEST_CLASS}")
    add_test(NAME schema-jni-unicode COMMAND "${SCHEMA_JAVA}" -cp "${CMAKE_CURRENT_BINARY_DIR}/jni-test" com.wludy.rolithax.launcher.shared.console.SchemaNativeBridge "$<TARGET_FILE:rolithax_schema>")
  endif()
endif()
install(TARGETS rolithax_schema RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/rolithax_schema.h DESTINATION include)
if(ROLITHAX_SCHEMA_CLI)
  install(TARGETS rolithax-schema RUNTIME DESTINATION bin)
endif()
